7   
8   import static ch.rasc.sse.eventbus.SseEvent.DEFAULT_EVENT;
9   
10  import ch.rasc.sse.eventbus.SseEvent;
11  import ch.rasc.sse.eventbus.SseEventBus;
12  import java.io.IOException;
13  import java.lang.invoke.MethodHandles;
14  import java.net.URI;
15  import java.net.URISyntaxException;
16  import java.time.Instant;
17  import java.time.OffsetDateTime;
18  import java.time.ZoneId;
19  import org.apache.commons.lang3.StringUtils;
20  import org.slf4j.Logger;
21  import org.slf4j.LoggerFactory;
22  import org.springframework.beans.factory.annotation.Value;
23  import org.springframework.stereotype.Service;
24  import org.tailormap.api.admin.model.CapabilitiesLoadingEvent;
25  import org.tailormap.api.admin.model.ServerSentEvent;
26  import org.tailormap.api.geotools.featuresources.JDBCFeatureSourceHelper;
27  import org.tailormap.api.geotools.featuresources.WFSFeatureSourceHelper;
28  import org.tailormap.api.persistence.TMFeatureSource;
29  import tools.jackson.core.JacksonException;
30  import tools.jackson.databind.json.JsonMapper;
31  
32  @Service
33  public class TMFeatureSourceHelper implements CapabilitiesLoadingProgressReporting {
34    private static final Logger logger =
35        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
36  
37    @Value("${tailormap-api.timeout}")
38    private int timeout;
39  
40    private final SseEventBus eventBus;
41    private final JsonMapper mapper;
42  
43    public TMFeatureSourceHelper(SseEventBus eventBus, JsonMapper mapper) {
44      this.eventBus = eventBus;
45      this.mapper = mapper;
46    }
47  
48    public TMFeatureSource loadCapabilities(TMFeatureSource featureSource) throws IOException {
49      final Instant startedAt = Instant.now();
50      final OffsetDateTime startedAtOffset =
51          startedAt.atOffset(ZoneId.systemDefault().getRules().getOffset(startedAt));
52      CapabilitiesLoadingEvent event = new CapabilitiesLoadingEvent()
53          .startedAt(startedAtOffset)
54          .title(
55              StringUtils.isNotBlank(featureSource.getTitle())
56                  ? featureSource.getTitle()
57                  : TMFeatureSource.Protocol.JDBC.equals(featureSource.getProtocol())
58                      ? String.valueOf(featureSource
59                          .getJdbcConnection()
60                          .getDatabase())
61                      : featureSource.getUrl())
62          .id(featureSource.getId() != null ? featureSource.getId().toString() : null);
63  
64      switch (featureSource.getProtocol()) {
65        case WFS -> new WFSFeatureSourceHelper().loadCapabilities(featureSource, timeout, this, event);
66        case JDBC -> new JDBCFeatureSourceHelper().loadCapabilities(featureSource, timeout, this, event);
67        default -> throw new UnsupportedOperationException("Unsupported protocol: " + featureSource.getProtocol());
68      }
69  
70      reportCapabilitiesLoadingProgress(event.message("Finished loading feature source capabilities in "
71              + (Instant.now().toEpochMilli() - startedAt.toEpochMilli()) + " ms")
72          .progress(event.getTotal()));
73      return featureSource;
74    }
75  
76    public TMFeatureSource createFeatureSource(TMFeatureSource featureSource)
77        throws IOException, IllegalArgumentException {
78      switch (featureSource.getProtocol()) {
79        case WFS -> {
80          if (StringUtils.isBlank(featureSource.getUrl())) {
81            throw new IllegalArgumentException("WFS feature source requires a URL");
82          }
83          URI uri;
84          try {
85            uri = new URI(featureSource.getUrl());
86          } catch (URISyntaxException | NullPointerException e) {
87            throw new IllegalArgumentException("Invalid URI");
88          }
89          if (!"https".equals(uri.getScheme()) && !"http".equals(uri.getScheme())) {
90            throw new IllegalArgumentException("Invalid URI scheme");
91          }
92        }
93        case JDBC -> {
94          if (featureSource.getJdbcConnection() == null) {
95            throw new IllegalArgumentException("JDBC connection properties are required");
96          }
97          if (featureSource.getAuthentication() == null) {
98            throw new IllegalArgumentException("Database username and password are required");
99          }
100       }
101     }
102 
103     return loadCapabilities(featureSource);
104   }
105 
106   @Override
107   public void reportCapabilitiesLoadingProgress(CapabilitiesLoadingEvent event) {
108     ServerSentEvent serverSentEvent = new ServerSentEvent()
109         .eventType(ServerSentEvent.EventTypeEnum.CAPABILITIES_LOADING)
110         .details(event);
111     try {
112       if (this.eventBus.countSubscribers(DEFAULT_EVENT) == 0) {
113         return;
114       }
115       this.eventBus.handleEvent(SseEvent.of(DEFAULT_EVENT, mapper.writeValueAsString(serverSentEvent)));
116     } catch (JacksonException e) {
117       logger.error("Error publishing feature source capabilities loading progress event", e);
118     }
119   }
120 }
