7   package org.tailormap.api.configuration;
8   
9   import java.sql.Date;
10  import org.springframework.boot.jackson.JacksonComponent;
11  import tools.jackson.core.JacksonException;
12  import tools.jackson.core.JsonGenerator;
13  import tools.jackson.databind.SerializationContext;
14  import tools.jackson.databind.ser.std.StdSerializer;
15  
16  /**
17   * Serializes a {@link java.sql.Date} as an ISO date without a time, for example {@code "2026-10-21"}.
18   *
19   * <p>GeoTools returns the values of {@code date} database columns as {@link java.sql.Date}, and the features API passes
20   * them to Jackson unchanged. Jackson 2 serialized a {@code java.sql.Date} with {@code toString()} as a date only.
21   * Jackson 3 serializes it like a {@link java.util.Date}, as a timestamp in UTC (for example
22   * {@code "2026-10-21T00:00:00.000Z"}), which the viewer does not accept for date fields. When the JVM time zone is not
23   * UTC that timestamp can even fall on the previous day. Jackson 3 has no feature to restore the old behaviour, so this
24   * serializer does.
25   *
26   * <p>{@link Date#toLocalDate()} uses the JVM default time zone, the same time zone the JDBC driver used to create the
27   * {@code java.sql.Date}, so the date is the date stored in the database.
28   */
29  @JacksonComponent
30  public class SqlDateSerializer extends StdSerializer<Date> {
31    public SqlDateSerializer() {
32      super(Date.class);
33    }
34  
35    @Override
36    public void serialize(Date value, JsonGenerator gen, SerializationContext provider) throws JacksonException {
37      gen.writeString(value.toLocalDate().toString());
38    }
39  }
