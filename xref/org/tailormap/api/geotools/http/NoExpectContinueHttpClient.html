6   package org.tailormap.api.geotools.http;
7   
8   import java.lang.reflect.Field;
9   import org.apache.hc.client5.http.config.ConnectionConfig;
10  import org.apache.hc.client5.http.config.RequestConfig;
11  import org.apache.hc.client5.http.impl.io.PoolingHttpClientConnectionManager;
12  import org.geotools.http.commons.MultithreadedHttpClient;
13  
14  /**
15   * GeoTools {@link MultithreadedHttpClient} that doesn't send an {@code Expect: 100-continue} header with POST requests.
16   *
17   * <p>{@link MultithreadedHttpClient} enables the {@code Expect: 100-continue} handshake for all requests. With it, the
18   * Apache HttpClient sends a POST request without its body first and waits for a {@code 100 Continue} response before
19   * sending the body, or sends the body anyway after a timeout of 3 seconds. Some servers (for example the PDOK WFS
20   * services at {@code service.pdok.nl}) only send the {@code 100 Continue} response after receiving the body, so every
21   * POST request (WFS GetFeature, and DescribeFeatureType for WFS 1.1.0) takes at least 3 seconds. The handshake only
22   * avoids sending a large body that the server rejects based on the headers alone, which doesn't matter for the small
23   * XML request bodies GeoTools sends.
24   *
25   * <p>Also fixes {@link #setConnectTimeout(int)}: {@link MultithreadedHttpClient} (GeoTools 35.2) only replaces its
26   * connection config field, which the connection manager has already copied in the constructor, so the connect timeout
27   * always stays at the initial 30 seconds.
28   *
29   * <p>{@link MultithreadedHttpClient} doesn't allow changing its request configuration or accessing its connection
30   * manager, so this uses reflection. If that fails because of a changed GeoTools implementation, the constructor throws
31   * an exception, so this is caught by the unit test when upgrading GeoTools.
32   */
33  public class NoExpectContinueHttpClient extends MultithreadedHttpClient {
34    private final Field connectionConfigField;
35    private final PoolingHttpClientConnectionManager connectionManager;
36  
37    public NoExpectContinueHttpClient() {
38      super();
39      try {
40        Field requestConfigField = getField("requestConfig");
41        RequestConfig requestConfig = (RequestConfig) requestConfigField.get(this);
42        requestConfigField.set(
43            this,
44            RequestConfig.copy(requestConfig)
45                .setExpectContinueEnabled(false)
46                .build());
47  
48        connectionConfigField = getField("connectionConfig");
49        connectionManager = (PoolingHttpClientConnectionManager)
50            getField("connectionManager").get(this);
51      } catch (ReflectiveOperationException | ClassCastException e) {
52        throw new IllegalStateException(
53            "Can't change the configuration of " + MultithreadedHttpClient.class.getName(), e);
54      }
55    }
56  
57    private static Field getField(String name) throws NoSuchFieldException {
58      Field field = MultithreadedHttpClient.class.getDeclaredField(name);
59      field.setAccessible(true);
60      return field;
61    }
62  
63    /** Sets the connect timeout in seconds, also for new connections of the connection manager. */
64    @Override
65    public void setConnectTimeout(int connectTimeout) {
66      super.setConnectTimeout(connectTimeout);
67      try {
68        connectionManager.setDefaultConnectionConfig((ConnectionConfig) connectionConfigField.get(this));
69      } catch (IllegalAccessException e) {
70        throw new IllegalStateException(e);
71      }
72    }
73  }
