6   package org.tailormap.api.geotools.http;
7   
8   import java.io.IOException;
9   import java.util.List;
10  import org.geotools.http.AbstractHTTPClientFactory;
11  import org.geotools.http.HTTPBehavior;
12  import org.geotools.http.HTTPClient;
13  import org.geotools.http.HTTPConnectionPooling;
14  import org.geotools.http.LoggingHTTPClient;
15  import org.geotools.http.SimpleHttpClient;
16  
17  /**
18   * GeoTools HTTP client factory that uses {@link NoExpectContinueHttpClient} instead of
19   * {@link org.geotools.http.commons.MultithreadedHttpClient}. Registered as a service in
20   * {@code META-INF/services/org.geotools.http.HTTPClientFactory} and made the default by setting the
21   * {@link org.geotools.util.factory.Hints#HTTP_CLIENT_FACTORY} hint in
22   * {@link org.tailormap.api.configuration.GeoToolsConfiguration}.
23   *
24   * <p>Without the hint, GeoTools creates a {@link SimpleHttpClient} (from
25   * {@link org.geotools.http.DefaultHTTPClientFactory}) unless a behavior such as {@link HTTPConnectionPooling} is
26   * requested, which the WFS datastore does. Because the hint makes this factory handle all requests, it does the same:
27   * it only creates the pooled {@link NoExpectContinueHttpClient} when {@link SimpleHttpClient} doesn't support a
28   * requested behavior. {@link SimpleHttpClient} doesn't send {@code Expect: 100-continue}, applies its connect timeout
29   * (the pooled client ignores {@code setConnectTimeout()} in GeoTools 35.2) and doesn't need to be closed, so code that
30   * uses the default client such as WMS/WMTS capabilities loading doesn't change.
31   */
32  public class NoExpectContinueHttpClientFactory extends AbstractHTTPClientFactory {
33    @Override
34    public List<Class<?>> clientClasses() {
35      return List.of(SimpleHttpClient.class, NoExpectContinueHttpClient.class);
36    }
37  
38    @Override
39    public HTTPClient createClient(List<Class<? extends HTTPBehavior>> behaviors) {
40      if (behaviors.stream().allMatch(behavior -> behavior.isAssignableFrom(SimpleHttpClient.class))) {
41        return new SimpleHttpClient();
42      }
43      return new NoExpectContinueHttpClient();
44    }
45  
46    @Override
47    protected HTTPClient createLogging(HTTPClient client) {
48      if (client instanceof HTTPConnectionPooling) {
49        // Keep HTTPConnectionPooling available when the HTTP_LOGGING hint is set: the WFS datastore casts to it
50        return new LoggingConnectionPoolingHTTPClient(client);
51      }
52      return super.createLogging(client);
53    }
54  
55    static class LoggingConnectionPoolingHTTPClient extends LoggingHTTPClient implements HTTPConnectionPooling {
56      LoggingConnectionPoolingHTTPClient(HTTPClient delegate) {
57        super(delegate);
58      }
59  
60      @Override
61      public int getMaxConnections() {
62        return ((HTTPConnectionPooling) delegate).getMaxConnections();
63      }
64  
65      @Override
66      public void setMaxConnections(int maxConnections) {
67        ((HTTPConnectionPooling) delegate).setMaxConnections(maxConnections);
68      }
69  
70      @Override
71      public void close() throws IOException {
72        ((HTTPConnectionPooling) delegate).close();
73      }
74    }
75  }
