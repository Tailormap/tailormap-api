6   package org.tailormap.api.geotools;
7   
8   import java.lang.invoke.MethodHandles;
9   import java.util.regex.Matcher;
10  import java.util.regex.Pattern;
11  import org.geotools.api.data.SimpleFeatureSource;
12  import org.geotools.api.filter.Filter;
13  import org.geotools.api.referencing.FactoryException;
14  import org.geotools.api.referencing.crs.CoordinateReferenceSystem;
15  import org.geotools.api.referencing.operation.MathTransform;
16  import org.geotools.filter.text.cql2.CQLException;
17  import org.geotools.filter.text.ecql.ECQL;
18  import org.geotools.referencing.CRS;
19  import org.jspecify.annotations.NonNull;
20  import org.slf4j.Logger;
21  import org.slf4j.LoggerFactory;
22  import org.tailormap.api.persistence.Application;
23  
24  public class FilterUtil {
25    private static final Pattern sridPattern = Pattern.compile("(?i)SRID=\\d+;");
26    private static final Logger logger =
27        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
28  
29    private FilterUtil() {
30      /* utility class */
31    }
32  
33    /**
34     * Parses a CQL filter string into a GeoTools Filter, transforming geometries to the data source CRS if
35     * needed/possible.
36     *
37     * @param filterCQL the CQL filter string
38     * @param application the application context
39     * @param featureSource the feature source
40     * @return the parsed filter, with geometries transformed to the data source CRS if needed
41     * @throws CQLException if the filter cannot be parsed
42     * @throws FactoryException if there is an error with the CRS
43     * @throws UnsupportedOperationException if the filter SRID does not match the application CRS or different SRID
44     *     appear in the filter
45     */
46    @NonNull public static Filter parseFilter(
47        @NonNull String filterCQL, @NonNull Application application, @NonNull SimpleFeatureSource featureSource)
48        throws CQLException, FactoryException, UnsupportedOperationException {
49      @SuppressWarnings("PMD.UseFilterUtil")
50      Filter filter = ECQL.toFilter(filterCQL);
51      // if the filter is spatial/has EWKT geometries and the SRID is different from the data source
52      // we need to transform the geometries to the data source CRS
53      String foundSrid = null;
54      final Matcher sridMatcher = sridPattern.matcher(filterCQL);
55      while (sridMatcher.find()) {
56        // only support the EPSG authority for now, so we replace the SRID= with EPSG: and remove the trailing ;
57        String srid = sridMatcher.group().replaceAll("(?i)SRID=", "EPSG:").replace(";", "");
58        if (foundSrid == null) {
59          foundSrid = srid;
60        } else if (!foundSrid.equalsIgnoreCase(srid)) {
61          throw new UnsupportedOperationException("All SRIDs in filter must be identical");
62        }
63      }
64      if (foundSrid != null) {
65        logger.trace("Filter contains SRID, checking if transformation is needed");
66        CoordinateReferenceSystem dataSourceCRS = featureSource.getSchema().getCoordinateReferenceSystem();
67        CoordinateReferenceSystem filterCRS = CRS.decode(foundSrid);
68        CoordinateReferenceSystem appCRS = CRS.decode(application.getCrs());
69  
70        if (!CRS.isEquivalent(appCRS, filterCRS)) {
71          // only support application CRS input
72          throw new UnsupportedOperationException("Filter SRID does not match application CRS");
73        }
74  
75        if (!CRS.isEquivalent(dataSourceCRS, filterCRS)) {
76          MathTransform transform = TransformationUtil.getTransformationToDataSource(application, featureSource);
77          if (transform != null) {
78            filter = TransformationUtil.transformFilterGeometries(filter, transform);
79          }
80        }
81      }
82      return filter;
83    }
84  }
