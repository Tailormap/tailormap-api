9   import java.net.URL;

12  import net.opengis.ows11.DomainType;

18  import org.geotools.http.HTTPClient;

21  /**
22   * {@link WFSClient} that lets a WFS 2.0.0 server sort, when the capabilities have the {@code ImplementsSorting}
23   * constraint. GeoTools 35.2 only lets a WFS 1.1.0 server sort.
24   *
25   * <p>When the client can't sort, {@code ContentFeatureSource} sorts each page of features in memory after getting it
26   * from the server with {@code STARTINDEX} and {@code COUNT}. Each page is sorted on its own, which gives a wrong order
27   * across pages, and it fails when a feature has an attribute that isn't {@code Serializable}, such as an
28   * {@code Instant} for a date. That also happens without a sort order, because {@code ContentFeatureSource} adds
29   * {@code NATURAL_ORDER} when paging.
30   *
31   * <p>For WFS 2.0.0 the {@link StrictWFS_2_0_Strategy} GeoTools picks is replaced with {@link SortingWFS_2_0_Strategy},
32   * which sends the sort order to the server. The client then always says it can sort, so GeoTools never sorts in memory:
33   * in memory sorting is wrong with paging, and paging is always used for the attribute list. Instead:
34   *
35   * <ul>
36   *   <li>Natural order (also the one {@code ContentFeatureSource} adds when paging) is not sent, so the server uses its
37   *       own order. WFS 2.0 (ISO 19142) 7.9.2.5.4.4 requires that order to be the same across executions of the same
38   *       query, so it is stable across pages.
39   *   <li>Sorting on an attribute when the capabilities don't have the {@code ImplementsSorting} constraint is an error,
40   *       see {@link SortingWFS_2_0_Strategy}.
41   * </ul>
42   *
43   * <p>Other strategies, such as {@code ArcGisStrategy_2_0}, are left to GeoTools: no sorting on the server, and each
44   * page sorted in memory.
45   *
46   * <p>This can be removed when GeoTools supports sorting for WFS 2.0.0 itself.
47   */
48  public class SortingWFSClient extends WFSClient {
49    private final boolean canSort;
50  
51    public SortingWFSClient(URL capabilitiesURL, HTTPClient httpClient, WFSConfig config)
52        throws IOException, ServiceException {
53      super(capabilitiesURL, httpClient, config);
54  
55      // The strategy is chosen in a private method in the WFSClient constructor, so replace it afterwards. Only
56      // replace the exact class, not subclasses such as ArcGisStrategy_2_0.
57      if (getStrategy().getClass() == StrictWFS_2_0_Strategy.class) {
58        SortingWFS_2_0_Strategy strategy = new SortingWFS_2_0_Strategy();
59        strategy.setConfig(config);
60        strategy.setCapabilities(getCapabilities());
61        strategy.setImplementsSorting(implementsSorting(getCapabilities()));
62        super.specification = strategy;
63        canSort = true;
64      } else {
65        canSort = super.canSort();
66      }
67    }
68  
69    /**
70     * Check whether WFS 2.0.0 capabilities have the {@code ImplementsSorting} conformance constraint set to
71     * {@code TRUE}. This is a constraint in {@code fes:Filter_Capabilities/fes:Conformance}, not in
72     * {@code ows:OperationsMetadata}.
73     *
74     * @param capabilities the capabilities
75     * @return whether the server implements sorting
76     */
77    static boolean implementsSorting(WFSGetCapabilities capabilities) {
78      if (!(capabilities.getParsedCapabilities() instanceof WFSCapabilitiesType caps)) {
79        return false;
80      }
81      FilterCapabilitiesType filterCapabilities = caps.getFilterCapabilities();
82      if (filterCapabilities == null || filterCapabilities.getConformance() == null) {
83        return false;
84      }
85      for (DomainType constraint : filterCapabilities.getConformance().getConstraint()) {
86        if ("ImplementsSorting".equals(constraint.getName())) {
87          return constraint.getDefaultValue() != null
88              && constraint.getDefaultValue().getValue() != null
89              && "TRUE"
90                  .equals(constraint
91                      .getDefaultValue()
92                      .getValue()
93                      .trim()
94                      .toUpperCase(Locale.ROOT));
95        }
96      }
97      return false;
98    }
99  
100   @Override
101   public boolean canSort() {
102     return canSort;
103   }
104 }
