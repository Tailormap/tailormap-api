8   import java.io.IOException;
9   import java.net.URL;
10  import java.util.Map;
11  import org.geotools.data.wfs.WFSDataStore;
12  import org.geotools.data.wfs.WFSDataStoreFactory;
13  import org.geotools.data.wfs.internal.WFSConfig;
14  import org.geotools.factory.CommonFactoryFinder;
15  import org.geotools.feature.type.FeatureTypeFactoryImpl;
16  import org.geotools.http.HTTPClient;
17  import org.geotools.ows.ServiceException;
18  import org.locationtech.jts.geom.GeometryFactory;
19  import org.locationtech.jts.geom.impl.PackedCoordinateSequenceFactory;

21  /**
22   * {@link WFSDataStoreFactory} that creates a {@link WFSDataStore} with a {@link SortingWFSClient}. This is a copy of
23   * {@link WFSDataStoreFactory#createDataStore(Map)} (GeoTools 35.2) with only the client class changed, because the
24   * client is created inline there.
25   *
26   * <p>This is not registered as a {@code DataStoreFactorySpi}, use it directly. It can be removed when GeoTools supports
27   * sorting for WFS 2.0.0 itself.
28   */
29  public class SortingWFSDataStoreFactory extends WFSDataStoreFactory {
30    @Override
31    public WFSDataStore createDataStore(final Map<String, ?> params) throws IOException {
32      final WFSConfig config = WFSConfig.fromParams(params);
33  
34      String user = config.getUser();
35      String password = config.getPassword();
36      if ((user == null) != (password == null)) {
37        throw new IOException("Cannot define only one of USERNAME or PASSWORD, must define both or neither");
38      }
39  
40      final URL capabilitiesURL = URL.lookUp(params);
41  
42      final HTTPClient http = getHttpClient(params);
43      http.setTryGzip(config.isTryGZIP());
44      http.setUser(user);
45      http.setPassword(password);
46      int timeoutMillis = config.getTimeoutMillis();
47      http.setConnectTimeout(timeoutMillis / 1000);
48      http.setReadTimeout(timeoutMillis / 1000);
49  
50      SortingWFSClient wfsClient;
51      try {
52        wfsClient = new SortingWFSClient(capabilitiesURL, http, config);
53      } catch (ServiceException e) {
54        throw new IOException(e);
55      }
56  
57      WFSDataStore dataStore = new WFSDataStore(wfsClient);
58      dataStore.setFilterFactory(CommonFactoryFinder.getFilterFactory(null));
59      dataStore.setGeometryFactory(new GeometryFactory(PackedCoordinateSequenceFactory.DOUBLE_FACTORY));
60      dataStore.setFeatureTypeFactory(new FeatureTypeFactoryImpl());
61      dataStore.setFeatureFactory(CommonFactoryFinder.getFeatureFactory(null));
62      dataStore.setDataStoreFactory(this);
63      dataStore.setNamespaceURI(config.getNamespaceOverride());
64      return dataStore;
65    }
66  }
