36  /**
37   * Counts features with a WFS {@code resultType=hits} GetFeature request when GeoTools does not.
38   *
39   * <p>GeoTools only does a hits request for WFS 1.x. For WFS 2.0.0 the {@code StrictWFS_2_0_Strategy} does not support
40   * hits (its pull parser can't read the count), so {@link SimpleFeatureSource#getCount(Query)} always returns -1 without
41   * asking the server. This class does the hits request itself, using the GeoTools request so the filter is encoded the
42   * same way as for the GetFeature request for the features, and reads the {@code numberMatched} attribute.
43   *
44   * <p>A hits request always counts, also on MapServer which returns {@code numberMatched="unknown"} in a GetFeature
45   * response with features (unless {@code wfs_compute_number_matched} is set). Counting can be slow for large layers.
46   */
47  public final class WFSHitsCountUtil {
48    private static final Logger logger =
49        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
50  
51    private WFSHitsCountUtil() {}
52  
53    /**
54     * Count the features of a feature source. Calls {@link SimpleFeatureSource#getCount(Query)} and if that returns -1
55     * for a WFS which GeoTools does not count with a hits request (WFS 2.0.0), does a hits request.
56     *
57     * @param fs the feature source, may be of any type
58     * @param query the query
59     * @return the number of features, or -1 if unknown
60     * @throws IOException when {@link SimpleFeatureSource#getCount(Query)} fails; a failing hits request returns -1
61     */
62    @SuppressWarnings("PMD.UseWFSHitsCountUtil")
63    public static int getCount(SimpleFeatureSource fs, Query query) throws IOException {
64      int count = fs.getCount(query);
65      if (count >= 0
66          || !(fs.getDataStore() instanceof WFSDataStore wfsDataStore)
67          || wfsDataStore.getWfsClient().canCount()) {
68        // When the client can count, GeoTools already did a hits request
69        return count;
70      }
71      try {
72        return getHitsCount(wfsDataStore, fs, query);
73      } catch (Exception e) {
74        logger.warn("WFS hits request to count features of {} failed: {}", fs.getName(), e.getMessage());
75        logger.debug("WFS hits request exception", e);
76        return -1;
77      }
78    }
79  
80    private static int getHitsCount(WFSDataStore wfsDataStore, SimpleFeatureSource fs, Query query)
81        throws IOException, XMLStreamException {
82      WFSClient client = wfsDataStore.getWfsClient();
83      QName remoteTypeName = wfsDataStore.getRemoteTypeName(fs.getName());
84      SimpleFeatureType remoteFeatureType = wfsDataStore.getRemoteSimpleFeatureType(remoteTypeName);
85  
86      Filter filter = query.getFilter() == null ? Filter.INCLUDE : query.getFilter();
87      Filter[] split = client.splitFilters(remoteTypeName, filter);
88      if (!Filter.INCLUDE.equals(split[1])) {
89        // Part of the filter must be evaluated after retrieving the features, the server can't count
90        logger.debug("Filter not fully supported by WFS, can't count with hits request: {}", filter);
91        return -1;
92      }
93      filter = invertAxisInFilterIfNeeded(client, split[0], query, remoteFeatureType);
94  
95      GetFeatureRequest request = client.createGetFeatureRequest();
96      request.setTypeName(remoteTypeName);
97      request.setFullType(remoteFeatureType);
98      request.setFilter(filter);
99      request.setResultType(ResultType.HITS);
100     request.setHints(query.getHints());
101 
102     HTTPClient http = client.getHTTPClient();
103     HTTPResponse response;
104     if (request.requiresPost()) {
105       ByteArrayOutputStream out = new ByteArrayOutputStream();
106       request.performPostOutput(out);
107       response = http.post(
108           request.getFinalURL(), new ByteArrayInputStream(out.toByteArray()), request.getPostContentType());
109     } else {
110       response = http.get(request.getFinalURL());
111     }
112     try (InputStream in = response.getResponseStream()) {
113       int count = parseNumberMatched(in);
114       logger.debug("WFS hits request for {} returned {} features", remoteTypeName, count);
115       return count;
116     } finally {
117       response.dispose();
118     }
119   }

121   /**
122    * Same as the private {@code WFSFeatureSource.invertAxisInFilterIfNeeded()} which GeoTools applies to the filter of
123    * a GetFeature request.
124    */
125   private static Filter invertAxisInFilterIfNeeded(
126       WFSClient client, Filter filter, Query query, SimpleFeatureType featureType) {
127     CoordinateReferenceSystem crs = query.getCoordinateSystem();
128     if (crs == null) {
129       crs = featureType.getCoordinateReferenceSystem();
130     }
131     if (WFSConfig.invertAxisNeeded(client.getAxisOrderFilter(), crs)) {
132       return (Filter) filter.accept(
133           new InvertAxisFilterVisitor(CommonFactoryFinder.getFilterFactory(null), new GeometryFactory()),
134           null);
135     }
136     return filter;
137   }

139   /**
140    * Read the number of features from a GetFeature hits response: the {@code numberMatched} attribute (WFS 2.0.0) or
141    * the {@code numberOfFeatures} attribute (WFS 1.x) of the root {@code FeatureCollection} element.
142    *
143    * @param in the response
144    * @return the number of features, or -1 if the response is not a feature collection (for example an exception
145    *     report) or the number is not a non-negative integer (for example {@code "unknown"})
146    * @throws XMLStreamException if the response is not XML
147    */
148   public static int parseNumberMatched(InputStream in) throws XMLStreamException {
149     XMLStreamReader reader = SecureXmlFactories.newXMLInputFactory().createXMLStreamReader(in);
150     try {
151       while (reader.hasNext()) {
152         if (reader.next() == XMLStreamConstants.START_ELEMENT) {
153           if (!"FeatureCollection".equals(reader.getLocalName())) {
154             logger.warn("Unexpected root element in WFS hits response: {}", reader.getName());
155             return -1;
156           }
157           String value = reader.getAttributeValue(null, "numberMatched");
158           if (value == null) {
159             value = reader.getAttributeValue(null, "numberOfFeatures");
160           }
161           return parseCount(value);
162         }
163       }
164       return -1;
165     } finally {
166       reader.close();
167     }
168   }
169 
170   private static int parseCount(String value) {
171     if (value == null) {
172       return -1;
173     }
174     try {
175       int count = Integer.parseInt(value.trim());
176       return count >= 0 ? count : -1;
177     } catch (NumberFormatException e) {
178       // "unknown"
179       return -1;
180     }
181   }
