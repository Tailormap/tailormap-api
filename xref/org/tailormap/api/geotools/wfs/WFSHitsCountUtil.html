45  /**
46   * Counts features with a WFS {@code resultType=hits} GetFeature request when GeoTools does not.
47   *
48   * <p>GeoTools only does a hits request for WFS 1.x. For WFS 2.0.0 the {@code StrictWFS_2_0_Strategy} does not support
49   * hits (its pull parser can't read the count), so {@link SimpleFeatureSource#getCount(Query)} always returns -1 without
50   * asking the server. This class does the hits request itself, using the GeoTools request so the filter is encoded the
51   * same way as for the GetFeature request for the features, and reads the {@code numberMatched} attribute.
52   *
53   * <p>A hits request always counts, also on MapServer which returns {@code numberMatched="unknown"} in a GetFeature
54   * response with features (unless {@code wfs_compute_number_matched} is set). Counting can be slow for large layers. See
55   * {@link #getCount(SimpleFeatureSource, Query, Duration, Duration)} for not waiting long for a slow count.
56   */
57  public final class WFSHitsCountUtil {
58    private static final Logger logger =
59        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
60  
61    /**
62     * WFS feature types (capabilities URL and type name) for which a hits request timed out, with the time until which
63     * no hits request is done. In memory only: a slow count is a property of the server and the size of the layer,
64     * which rarely changes, and after a restart it is only one hits request to find out again.
65     *
66     * <p>Remembered separately per kind of filter ({@link FilterKind}): counting can be quick without a filter and slow
67     * with one, or the other way around. For example a spatial filter on GeoServer/PostGIS which doesn't use the
68     * spatial index, or a slow total for a large layer with a quick filter on an indexed column.
69     */
70    private static final Map<String, Instant> slowHits = new ConcurrentHashMap<>();
71  
72    /** The kind of filter of a count, to remember slow hits requests separately per kind. */
73    enum FilterKind {
74      NONE,
75      ATTRIBUTE,
76      /** A filter with a spatial part, also when it has attribute parts: the spatial part is the likely slow one. */
77      SPATIAL
78    }
79  
80    private WFSHitsCountUtil() {}
81  
82    /**
83     * Count the features of a feature source. Calls {@link SimpleFeatureSource#getCount(Query)} and if that returns -1
84     * for a WFS which GeoTools does not count with a hits request (WFS 2.0.0), does a hits request.
85     *
86     * @param fs the feature source, may be of any type
87     * @param query the query
88     * @return the number of features, or -1 if unknown
89     * @throws IOException when {@link SimpleFeatureSource#getCount(Query)} fails; a failing hits request returns -1
90     */
91    public static int getCount(SimpleFeatureSource fs, Query query) throws IOException {
92      return countWithHits(fs, query, null, null);
93    }
94  
95    /**
96     * Count the features of a feature source like {@link #getCount(SimpleFeatureSource, Query)}, but without waiting
97     * long for a slow hits request. For when the count is nice to have, such as the total for a page of features.
98     *
99     * <p>A hits request which takes longer than {@code hitsTimeout} is aborted and the feature type is remembered as
100    * slow, so it isn't counted with a hits request for {@code slowHitsTtl} and the count is -1 without waiting. Some
101    * servers count very slowly for large layers (PDOK NWB wegvakken: ~18 s for a hits request versus 0.1 s for a page
102    * of features), while counting is quick for most.
103    *
104    * @param fs the feature source, may be of any type
105    * @param query the query
106    * @param hitsTimeout how long to wait for a hits request, in whole seconds (at least 1 s, the HTTP read timeout)
107    * @param slowHitsTtl how long not to count a feature type after its hits request timed out
108    * @return the number of features, or -1 if unknown or too slow to count
109    * @throws IOException when {@link SimpleFeatureSource#getCount(Query)} fails; a failing hits request returns -1
110    */
111   public static int getCount(SimpleFeatureSource fs, Query query, Duration hitsTimeout, Duration slowHitsTtl)
112       throws IOException {
113     return countWithHits(fs, query, hitsTimeout, slowHitsTtl);
114   }
115 
116   /** Without {@code hitsTimeout} the normal read timeout is used and slow feature types are not remembered. */
117   @SuppressWarnings("PMD.UseWFSHitsCountUtil")
118   private static int countWithHits(
119       SimpleFeatureSource fs, Query query, @Nullable Duration hitsTimeout, @Nullable Duration slowHitsTtl)
120       throws IOException {
121     int count = fs.getCount(query);
122     if (count >= 0
123         || !(fs.getDataStore() instanceof WFSDataStore wfsDataStore)
124         || wfsDataStore.getWfsClient().canCount()) {
125       // When the client can count, GeoTools already did a hits request
126       return count;
127     }
128     String slowHitsKey = null;
129     if (hitsTimeout != null && slowHitsTtl != null) {
130       slowHitsKey = wfsDataStore.getInfo().getSource()
131           + "|"
132           + wfsDataStore.getRemoteTypeName(fs.getName())
133           + "|"
134           + getFilterKind(query.getFilter(), fs.getSchema());
135       if (isSlowHits(slowHitsKey)) {
136         logger.debug("Not counting features of {}, an earlier hits request timed out", slowHitsKey);
137         return -1;
138       }
139     }
140     try {
141       return getHitsCount(wfsDataStore, fs, query, hitsTimeout);
142     } catch (Exception e) {
143       if (slowHitsKey != null && ExceptionUtils.indexOfType(e, SocketTimeoutException.class) >= 0) {
144         logger.info(
145             "WFS hits request to count features of {} took longer than {}, not counting for {}",
146             slowHitsKey,
147             hitsTimeout,
148             slowHitsTtl);
149         slowHits.put(slowHitsKey, Instant.now().plus(slowHitsTtl));
150         return -1;
151       }
152       logger.warn("WFS hits request to count features of {} failed: {}", fs.getName(), e.getMessage());
153       logger.debug("WFS hits request exception", e);
154       return -1;
155     }
156   }

158   /**
159    * The kind of filter. A filter is spatial when it uses a geometry attribute of the feature type, normally with a
160    * spatial operator. Checks the attributes instead of the operators, so that all spatial operators are covered.
161    */
162   static FilterKind getFilterKind(@Nullable Filter filter, SimpleFeatureType schema) {
163     if (filter == null || Filter.INCLUDE.equals(filter)) {
164       return FilterKind.NONE;
165     }
166     FilterAttributeExtractor extractor = new FilterAttributeExtractor(schema);
167     filter.accept(extractor, null);
168     return extractor.getAttributeNameSet().stream()
169             .anyMatch(name -> schema.getDescriptor(name) instanceof GeometryDescriptor)
170         ? FilterKind.SPATIAL
171         : FilterKind.ATTRIBUTE;
172   }

174   private static boolean isSlowHits(String slowHitsKey) {
175     Instant until = slowHits.get(slowHitsKey);
176     if (until == null) {
177       return false;
178     }
179     if (until.isBefore(Instant.now())) {
180       slowHits.remove(slowHitsKey, until);
181       return false;
182     }
183     return true;
184   }

186   /** Forget the feature types for which a hits request timed out. For tests. */
187   static void clearSlowHits() {
188     slowHits.clear();
189   }

191   private static int getHitsCount(
192       WFSDataStore wfsDataStore, SimpleFeatureSource fs, Query query, @Nullable Duration hitsTimeout)
193       throws IOException, XMLStreamException {
194     WFSClient client = wfsDataStore.getWfsClient();
195     QName remoteTypeName = wfsDataStore.getRemoteTypeName(fs.getName());
196     SimpleFeatureType remoteFeatureType = wfsDataStore.getRemoteSimpleFeatureType(remoteTypeName);
197 
198     Filter filter = query.getFilter() == null ? Filter.INCLUDE : query.getFilter();
199     Filter[] split = client.splitFilters(remoteTypeName, filter);
200     if (!Filter.INCLUDE.equals(split[1])) {
201       // Part of the filter must be evaluated after retrieving the features, the server can't count
202       logger.debug("Filter not fully supported by WFS, can't count with hits request: {}", filter);
203       return -1;
204     }
205     filter = invertAxisInFilterIfNeeded(client, split[0], query, remoteFeatureType);
206 
207     GetFeatureRequest request = client.createGetFeatureRequest();
208     request.setTypeName(remoteTypeName);
209     request.setFullType(remoteFeatureType);
210     request.setFilter(filter);
211     request.setResultType(ResultType.HITS);
212     request.setHints(query.getHints());
213 
214     // The data store and its HTTP client are not shared between threads (see FeatureSourceFactoryHelper), so the
215     // read timeout can be changed for this request only
216     HTTPClient http = client.getHTTPClient();
217     int readTimeout = http.getReadTimeout();
218     if (hitsTimeout != null) {
219       http.setReadTimeout((int) Math.max(1, hitsTimeout.toSeconds()));
220     }
221     try {
222       HTTPResponse response;
223       if (request.requiresPost()) {
224         ByteArrayOutputStream out = new ByteArrayOutputStream();
225         request.performPostOutput(out);
226         response = http.post(
227             request.getFinalURL(),
228             new ByteArrayInputStream(out.toByteArray()),
229             request.getPostContentType());
230       } else {
231         response = http.get(request.getFinalURL());
232       }
233       try (InputStream in = response.getResponseStream()) {
234         int count = parseNumberMatched(in);
235         logger.debug("WFS hits request for {} returned {} features", remoteTypeName, count);
236         return count;
237       } finally {
238         response.dispose();
239       }
240     } finally {
241       http.setReadTimeout(readTimeout);
242     }
243   }

245   /**
246    * Same as the private {@code WFSFeatureSource.invertAxisInFilterIfNeeded()} which GeoTools applies to the filter of
247    * a GetFeature request.
248    */
249   private static Filter invertAxisInFilterIfNeeded(
250       WFSClient client, Filter filter, Query query, SimpleFeatureType featureType) {
251     CoordinateReferenceSystem crs = query.getCoordinateSystem();
252     if (crs == null) {
253       crs = featureType.getCoordinateReferenceSystem();
254     }
255     if (WFSConfig.invertAxisNeeded(client.getAxisOrderFilter(), crs)) {
256       return (Filter) filter.accept(
257           new InvertAxisFilterVisitor(CommonFactoryFinder.getFilterFactory(null), new GeometryFactory()),
258           null);
259     }
260     return filter;
261   }

263   /**
264    * Read the number of features from a GetFeature hits response: the {@code numberMatched} attribute (WFS 2.0.0) or
265    * the {@code numberOfFeatures} attribute (WFS 1.x) of the root {@code FeatureCollection} element.
266    *
267    * @param in the response
268    * @return the number of features, or -1 if the response is not a feature collection (for example an exception
269    *     report) or the number is not a non-negative integer (for example {@code "unknown"})
270    * @throws XMLStreamException if the response is not XML
271    */
272   public static int parseNumberMatched(InputStream in) throws XMLStreamException {
273     XMLStreamReader reader = SecureXmlFactories.newXMLInputFactory().createXMLStreamReader(in);
274     try {
275       while (reader.hasNext()) {
276         if (reader.next() == XMLStreamConstants.START_ELEMENT) {
277           if (!"FeatureCollection".equals(reader.getLocalName())) {
278             logger.warn("Unexpected root element in WFS hits response: {}", reader.getName());
279             return -1;
280           }
281           String value = reader.getAttributeValue(null, "numberMatched");
282           if (value == null) {
283             value = reader.getAttributeValue(null, "numberOfFeatures");
284           }
285           return parseCount(value);
286         }
287       }
288       return -1;
289     } finally {
290       reader.close();
291     }
292   }
293 
294   private static int parseCount(String value) {
295     if (value == null) {
296       return -1;
297     }
298     try {
299       int count = Integer.parseInt(value.trim());
300       return count >= 0 ? count : -1;
301     } catch (NumberFormatException e) {
302       // "unknown"
303       return -1;
304     }
305   }
