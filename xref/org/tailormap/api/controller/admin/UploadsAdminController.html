16  import java.util.stream.Stream;
17  import org.slf4j.Logger;
18  import org.slf4j.LoggerFactory;
19  import org.springframework.http.HttpStatus;
20  import org.springframework.http.MediaType;
21  import org.springframework.http.ResponseEntity;
22  import org.springframework.transaction.annotation.Transactional;
23  import org.springframework.web.bind.annotation.DeleteMapping;
24  import org.springframework.web.bind.annotation.ExceptionHandler;

27  import org.springframework.web.bind.annotation.PostMapping;

30  import org.tailormap.api.persistence.Upload;

34  import org.tailormap.api.service.ZipService;
35  import org.tailormap.api.viewer.model.ErrorResponse;
36  
37  @RestController
38  public class UploadsAdminController {
39    private static final Logger logger =
40        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
41    private final UploadRepository uploadRepository;
42    private final ZipService zipService;
43  
44    public UploadsAdminController(UploadRepository uploadRepository, ZipService zipService) {
45      this.uploadRepository = uploadRepository;
46      this.zipService = zipService;
47    }
48  
49    @ExceptionHandler({IOException.class})
50    public ResponseEntity<?> handleException(Exception ex) {
51      // wrap the exception in a proper json response
52      return ResponseEntity.internalServerError()
53          .contentType(MediaType.APPLICATION_JSON)
54          .body(new ErrorResponse()
55              .message(
56                  ex.getMessage() != null
57                      ? ex.getMessage()
58                      : HttpStatus.INTERNAL_SERVER_ERROR.getReasonPhrase())
59              .code(HttpStatus.INTERNAL_SERVER_ERROR.value()));
60    }
61  
62    @ExceptionHandler(IllegalArgumentException.class)
63    public ResponseEntity<?> handleIllegalArgumentException(IllegalArgumentException ex) {
64      // wrap the exception in a proper json response
65      return ResponseEntity.badRequest()
66          .contentType(MediaType.APPLICATION_JSON)
67          .body(new ErrorResponse()
68              .message(ex.getMessage() != null ? ex.getMessage() : HttpStatus.BAD_REQUEST.getReasonPhrase())
69              .code(HttpStatus.BAD_REQUEST.value()));
70    }
71  
72    @PostMapping(
73        path = "${tailormap-api.admin.base-path}/uploads/find-by-hash/{category}",
74        consumes = "application/json")
75    public List<UploadMatch> findUploadsByHash(
76        @PathVariable UploadCategory category, @RequestBody List<String> hashes) {
77      return uploadRepository.findByHashIn(category, hashes);
78    }
79  
80    @Transactional(readOnly = true)
81    @GetMapping(path = "${tailormap-api.admin.base-path}/uploads/{uuids}", produces = "application/zip")
82    public byte[] downloadUploadsByCategory(@PathVariable("uuids") List<UUID> uuids) throws IOException {
83      // Authorisation check isn't needed: only admins are allowed on the admin base path
84      Path tempDir = Files.createTempDirectory("admin-uploads-");
85      try {
86        for (Upload upload : uploadRepository.findAllWithContentByIdIn(uuids)) {
87          // validate/sanitise filename: no directories allowed, only the filename itself
88          String safeFilename =
89              Path.of(upload.getFilename()).getFileName().toString();
90          Path filePath = tempDir.resolve(safeFilename);
91          Files.write(filePath, upload.getContent());
92        }
93  
94        Path zipFile = Files.createTempFile("admin-uploads-", ".zip");
95        logger.info("Created zip file {}", zipFile.toAbsolutePath());
96  
97        try {
98          zipService.zipDirectory(tempDir, zipFile);
99          return Files.readAllBytes(zipFile);
100       } finally {
101         Files.deleteIfExists(zipFile);
102       }
103     } finally {
104       try (Stream<Path> pathStream = Files.walk(tempDir)) {
105         pathStream.sorted(Comparator.reverseOrder()).forEach(path -> {
106           try {
107             Files.delete(path);
108           } catch (IOException e) {
109             // Ignore
110           }
111         });
112       }
113     }
114   }
115 
116   @DeleteMapping(path = "${tailormap-api.admin.base-path}/uploads/{uuids}")
117   public void deleteUploadsByCategory(@PathVariable("uuids") List<UUID> uuids) throws IllegalArgumentException {
118     uploadRepository.deleteAllById(uuids);
119   }
120 }
