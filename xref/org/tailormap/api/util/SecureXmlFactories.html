6   package org.tailormap.api.util;
7   
8   import javax.xml.XMLConstants;
9   import javax.xml.parsers.DocumentBuilderFactory;
10  import javax.xml.parsers.ParserConfigurationException;
11  import javax.xml.stream.XMLInputFactory;
12  
13  /**
14   * Factories for parsing XML hardened against XXE, for XML from external sources such as responses of OGC services. Use
15   * these instead of the JAXP {@code newInstance()} methods, enforced by the PMD rule {@code UseSecureXmlFactories}.
16   *
17   * <p>Only the JDK XML implementations are on the classpath, so no implementation-specific hardening is needed. A
18   * DOCTYPE is allowed because OGC responses can have one (WMS 1.1.1 exception reports usually do), but the external DTD
19   * and external entities are never loaded.
20   */
21  public final class SecureXmlFactories {
22    private SecureXmlFactories() {}
23  
24    /**
25     * Create a {@link DocumentBuilderFactory} which does not load external DTDs, entities or schemas.
26     *
27     * @return the hardened factory
28     * @throws ParserConfigurationException if the implementation does not support a feature
29     */
30    @SuppressWarnings("PMD.UseSecureXmlFactories")
31    public static DocumentBuilderFactory newDocumentBuilderFactory() throws ParserConfigurationException {
32      DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
33      factory.setFeature(XMLConstants.FEATURE_SECURE_PROCESSING, true);
34      factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
35      factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
36      factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);
37      factory.setAttribute(XMLConstants.ACCESS_EXTERNAL_DTD, "");
38      factory.setAttribute(XMLConstants.ACCESS_EXTERNAL_SCHEMA, "");
39      factory.setXIncludeAware(false);
40      factory.setExpandEntityReferences(false);
41      return factory;
42    }
43  
44    /**
45     * Create an {@link XMLInputFactory} (StAX) which does not process DTDs or external entities.
46     *
47     * @return the hardened factory
48     */
49    @SuppressWarnings("PMD.UseSecureXmlFactories")
50    public static XMLInputFactory newXMLInputFactory() {
51      XMLInputFactory factory = XMLInputFactory.newInstance();
52      factory.setProperty(XMLInputFactory.SUPPORT_DTD, false);
53      factory.setProperty(XMLInputFactory.IS_SUPPORTING_EXTERNAL_ENTITIES, false);
54      return factory;
55    }
56  }
