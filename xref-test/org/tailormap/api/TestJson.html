6   package org.tailormap.api;
7   
8   import tools.jackson.databind.json.JsonMapper;
9   
10  /**
11   * Build JSON for tests from basic Java structures such as {@code Map.of(...)} and {@code List.of(...)} instead of JSON
12   * strings, so values do not need to be escaped or concatenated and comparisons do not depend on formatting.
13   */
14  public final class TestJson {
15    private TestJson() {}
16  
17    /**
18     * Serialize a value to JSON.
19     *
20     * @param value a basic Java structure (for example nested maps and lists) or a model object
21     * @return the JSON string
22     */
23    public static String json(Object value) {
24      return JsonMapper.shared().writeValueAsString(value);
25    }
26  }
