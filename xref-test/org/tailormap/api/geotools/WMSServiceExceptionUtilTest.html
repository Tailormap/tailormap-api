8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   import static org.junit.jupiter.api.Assertions.assertFalse;
10  import static org.junit.jupiter.api.Assertions.assertNull;
11  
12  import java.nio.charset.StandardCharsets;
13  import java.nio.file.Files;
14  import java.nio.file.Path;
15  import mockwebserver3.MockResponse;
16  import mockwebserver3.MockWebServer;
17  import org.junit.jupiter.api.Test;
18  import org.junit.jupiter.api.io.TempDir;
19  
20  class WMSServiceExceptionUtilTest {
21    private static String getMessage(String xml) {
22      return WMSServiceExceptionUtil.tryGetServiceExceptionMessage(xml.getBytes(StandardCharsets.UTF_8));
23    }
24  
25    @Test
26    void wms_1_3_0_service_exception() {
27      assertEquals("code: LayerNotDefined: locator: layers: Could not find layer roads", getMessage("""
28  <?xml version="1.0" encoding="UTF-8"?>
29  <ServiceExceptionReport version="1.3.0" xmlns="http://www.opengis.net/ogc">
30  <ServiceException code="LayerNotDefined" locator="layers">
31  Could not find layer roads
32  </ServiceException>
33  </ServiceExceptionReport>
34  """));
35    }
36  
37    @Test
38    void ows_exception_report() {
39      assertEquals("code: NoApplicableCode: Error", getMessage("""
40  <?xml version="1.0" encoding="UTF-8"?>
41  <ows:ExceptionReport xmlns:ows="http://www.opengis.net/ows/1.1" version="2.0.0">
42  <ows:Exception exceptionCode="NoApplicableCode"><ows:ExceptionText>Error</ows:ExceptionText></ows:Exception>
43  </ows:ExceptionReport>
44  """));
45    }
46  
47    @Test
48    void not_xml() {
49      assertNull(getMessage("Internal Server Error"));
50    }
51  
52    @Test
53    void wms_1_1_1_with_doctype_does_not_load_external_dtd() throws Exception {
54      try (MockWebServer server = new MockWebServer()) {
55        server.enqueue(
56            new MockResponse.Builder().body("<!ENTITY e \"from DTD\">").build());
57        server.start();
58        String dtd = server.url("/exception_1_1_1.dtd").toString();
59        assertEquals("Error", getMessage("""
60  <?xml version="1.0" encoding="UTF-8"?>
61  <!DOCTYPE ServiceExceptionReport SYSTEM "%s">
62  <ServiceExceptionReport version="1.1.1"><ServiceException>Error</ServiceException></ServiceExceptionReport>
63  """.formatted(dtd)));
64        assertEquals(0, server.getRequestCount());
65      }
66    }
67  
68    @Test
69    void does_not_resolve_external_entity(@TempDir Path tempDir) throws Exception {
70      Path secret = tempDir.resolve("secret.txt");
71      Files.writeString(secret, "secret");
72      String message = getMessage("""
73  <?xml version="1.0" encoding="UTF-8"?>
74  <!DOCTYPE ServiceExceptionReport [<!ENTITY e SYSTEM "%s">]>
75  <ServiceExceptionReport version="1.3.0"><ServiceException>Error &e;</ServiceException></ServiceExceptionReport>
76  """.formatted(secret.toUri()));
77      assertFalse(message != null && message.contains("secret"), message);
78    }
79  
80    @Test
81    void does_not_load_external_parameter_entity() throws Exception {
82      try (MockWebServer server = new MockWebServer()) {
83        server.enqueue(
84            new MockResponse.Builder().body("<!ENTITY e \"from DTD\">").build());
85        server.start();
86        String url = server.url("/entities.dtd").toString();
87        getMessage("""
88  <?xml version="1.0" encoding="UTF-8"?>
89  <!DOCTYPE ServiceExceptionReport [<!ENTITY %% p SYSTEM "%s"> %%p;]>
90  <ServiceExceptionReport version="1.3.0"><ServiceException>Error &e;</ServiceException></ServiceExceptionReport>
91  """.formatted(url));
92        assertEquals(0, server.getRequestCount());
93      }
94    }
95  }
