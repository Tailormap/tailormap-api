200   /** Serves a WFS and records the GetFeature requests. */
201   private static class WFSDispatcher extends Dispatcher {
202     private final String capabilities;
203     private final Function<String, MockResponse> getFeatureResponse;
204     private final List<Map<String, String>> getFeatureRequests = new ArrayList<>();
205     private String url;
206 
207     WFSDispatcher(String capabilities, Function<String, MockResponse> getFeatureResponse) {
208       this.capabilities = capabilities;
209       this.getFeatureResponse = getFeatureResponse;
210     }
211 
212     @Override
213     public MockResponse dispatch(RecordedRequest request) {
214       Map<String, String> params = new HashMap<>();
215       for (String name : request.getUrl().queryParameterNames()) {
216         params.put(name.toUpperCase(Locale.ROOT), request.getUrl().queryParameter(name));
217       }
218       String xml = "application/xml";
219       return switch (params.getOrDefault("REQUEST", "")) {
220         case "GetCapabilities" -> response(xml, capabilities.replace("{url}", url));
221         case "DescribeFeatureType" -> response(xml, DESCRIBE_FEATURE_TYPE);
222         case "GetFeature" -> {
223           getFeatureRequests.add(params);
224           yield getFeatureResponse.apply(params.get("RESULTTYPE"));
225         }
226         default -> new MockResponse.Builder().code(400).build();
227       };
228     }
229   }
230 
231   private static MockResponse response(String contentType, String body) {
232     return new MockResponse.Builder()
233         .headers(new Headers(new String[] {"Content-Type", contentType}))
234         .body(body)
235         .build();
236   }
237 
238   private record CountResult(int count, List<Map<String, String>> getFeatureRequests) {}
239 
240   private static CountResult countWithServer(
241       String capabilities, Function<String, MockResponse> getFeatureResponse, String version, Query query)
242       throws IOException {
243     try (WFSDispatcher dispatcher = new WFSDispatcher(capabilities, getFeatureResponse);
244         MockWebServer server = new MockWebServer()) {
245       server.setDispatcher(dispatcher);
246       server.start();
247       dispatcher.url = server.url("/wfs").toString();
248       Map<String, Object> params = new HashMap<>();
249       params.put(
250           WFSDataStoreFactory.URL.key,
251           SimpleWFSHelper.getWFSRequestURL(dispatcher.url, "GetCapabilities", version, null)
252               .toURL());
253       params.put(WFSDataStoreFactory.PROTOCOL.key, false);
254       DataStore ds = new WFSDataStoreFactory().createDataStore(params);
255       try {
256         SimpleFeatureSource fs = ds.getFeatureSource(ds.getTypeNames()[0]);
257         return new CountResult(WFSHitsCountUtil.getCount(fs, query), dispatcher.getFeatureRequests);
258       } finally {
259         ds.dispose();
260       }
261     }
262   }

264   @Test
265   void hits_request_for_wfs_2_0_0() throws Exception {
266 
267     CountResult result = countWithServer(
268         CAPABILITIES_2_0_0,
269         resultType -> response("application/gml+xml; version=3.2", HITS_2_0_0.formatted("342")),
270         "2.0.0",
271         Query.ALL);
272     assertEquals(342, result.count());
273     assertEquals(1, result.getFeatureRequests().size());
274     Map<String, String> request = result.getFeatureRequests().getFirst();
275     assertEquals("HITS", request.get("RESULTTYPE").toUpperCase(Locale.ROOT));
276     assertEquals(TYPE_NAME, request.get("TYPENAMES"));
277   }

279   @Test
280   void hits_request_for_wfs_2_0_0_with_filter() throws Exception {
281     Query query = new Query(TYPE_NAME, FF.equals(FF.property("name"), FF.literal("A1")));
282     CountResult result = countWithServer(
283         CAPABILITIES_2_0_0,
284         resultType -> response("application/gml+xml; version=3.2", HITS_2_0_0.formatted("12")),
285         "2.0.0",
286         query);
287     assertEquals(12, result.count());
288     assertEquals(1, result.getFeatureRequests().size());
289     String filter = result.getFeatureRequests().getFirst().get("FILTER");
290     assertTrue(filter.contains("PropertyIsEqualTo") && filter.contains("A1"), filter);
291   }
