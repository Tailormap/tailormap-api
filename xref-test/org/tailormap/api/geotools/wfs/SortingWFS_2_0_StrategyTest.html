6   package org.tailormap.api.geotools.wfs;
7   
8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   import static org.junit.jupiter.api.Assertions.assertThrows;
10  import static org.junit.jupiter.api.Assertions.assertTrue;
11  
12  import java.util.List;
13  import org.geotools.api.filter.FilterFactory;
14  import org.geotools.api.filter.sort.SortBy;
15  import org.geotools.api.filter.sort.SortOrder;
16  import org.geotools.factory.CommonFactoryFinder;
17  import org.junit.jupiter.api.Test;
18  
19  class SortingWFS_2_0_StrategyTest {
20    private static final FilterFactory ff = CommonFactoryFinder.getFilterFactory();
21  
22    private static SortBy[] sortBy(SortBy... sortBy) {
23      return sortBy;
24    }
25  
26    @Test
27    void server_sort_by_without_natural_order() {
28      SortBy code = ff.sort("code", SortOrder.ASCENDING);
29      assertEquals(List.of(code), SortingWFS_2_0_Strategy.getServerSortBy(sortBy(SortBy.NATURAL_ORDER, code), true));
30      assertTrue(SortingWFS_2_0_Strategy.getServerSortBy(sortBy(SortBy.NATURAL_ORDER), true)
31          .isEmpty());
32      assertTrue(SortingWFS_2_0_Strategy.getServerSortBy(null, true).isEmpty());
33    }
34  
35    @Test
36    void natural_order_without_implements_sorting() {
37      assertTrue(SortingWFS_2_0_Strategy.getServerSortBy(sortBy(SortBy.NATURAL_ORDER), false)
38          .isEmpty());
39      assertTrue(SortingWFS_2_0_Strategy.getServerSortBy(null, false).isEmpty());
40    }
41  
42    @Test
43    void sort_on_attribute_without_implements_sorting() {
44      SortBy[] sortBy = sortBy(SortBy.NATURAL_ORDER, ff.sort("code", SortOrder.DESCENDING));
45      IllegalArgumentException e = assertThrows(
46          IllegalArgumentException.class, () -> SortingWFS_2_0_Strategy.getServerSortBy(sortBy, false));
47      assertEquals("This WFS does not support sorting", e.getMessage());
48    }
49  
50    @Test
51    void reverse_order() {
52      assertThrows(
53          IllegalArgumentException.class,
54          () -> SortingWFS_2_0_Strategy.getServerSortBy(sortBy(SortBy.REVERSE_ORDER), true));
55    }
56  
57    @Test
58    void sort_by_parameter() {
59      assertEquals(
60          "naam ASC", SortingWFS_2_0_Strategy.toSortByParameter(List.of(ff.sort("naam", SortOrder.ASCENDING))));
61      assertEquals(
62          "naam DESC,code ASC",
63          SortingWFS_2_0_Strategy.toSortByParameter(
64              List.of(ff.sort("naam", SortOrder.DESCENDING), ff.sort("code", SortOrder.ASCENDING))));
65    }
66  }
