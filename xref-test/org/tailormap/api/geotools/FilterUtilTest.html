36  class FilterUtilTest {

38    // Netherlands bounding box in RD New (EPSG:28992)
39    private static final String RD_NL_POLYGON =
40        "POLYGON((10000 306000, 280000 306000, 280000 640000, 10000 640000, 10000 306000))";
41  
42    private static SimpleFeatureSource RDFeatureSource;
43  
44    private static final Application RD_NL_APPLICATION = mock(Application.class, invocation -> {
45      if ("getCrs".equals(invocation.getMethod().getName())) {
46        return "EPSG:28992";
47      }
48      return RETURNS_DEFAULTS.answer(invocation);
49    });
50  
51    @BeforeAll
52    static void setupMocks() throws FactoryException {
53      RDFeatureSource = mockFeatureSource("EPSG:28992");
54    }
55  
56    private static SimpleFeatureSource mockFeatureSource(String crs) throws FactoryException {
57      SimpleFeatureSource featureSource = mock(SimpleFeatureSource.class);
58      SimpleFeatureType schema = mock(SimpleFeatureType.class);
59      when(featureSource.getSchema()).thenReturn(schema);
60      when(schema.getCoordinateReferenceSystem()).thenReturn(CRS.decode(crs));
61      return featureSource;
62    }
63  
64    @Test
65    void non_spatial_filter_returns_parsed_filter() throws CQLException, FactoryException {
66      Filter filter = FilterUtil.parseFilter("name = 'test'", RD_NL_APPLICATION, RDFeatureSource);
67      assertNotNull(filter);
68      assertInstanceOf(IsEqualsToImpl.class, filter);
69      assertEquals("[ name = test ]", filter.toString());
70    }
71  
72    @Test
73    void spatial_filter_same_crs_everywhere_no_transform() throws CQLException, FactoryException {
74      // filter SRID = app CRS = data source CRS → no transformation should occur
75      String cql = "INTERSECTS(geometry, SRID=28992;" + RD_NL_POLYGON + ")";
76      Filter filter = FilterUtil.parseFilter(cql, RD_NL_APPLICATION, RDFeatureSource);
77  
78      assertNotNull(filter);
79      assertInstanceOf(Intersects.class, filter);
80      // geometry stays in RD New, so X coordinate is >> 10
81      Geometry geom = (Geometry) ((Intersects) filter).getExpression2().evaluate(null);
82      assertNotNull(geom);
83      assertTrue(geom.getCentroid().getX() > 10000, "X should be in RD New range (not WGS84 longitude)");
84    }
85  
86    @Test
87    void spatial_filter_app_crs_matches_filter_srid_data_source_crs_differs_transform_applied()
88        throws CQLException, FactoryException {
89      // filter SRID = app CRS = EPSG:28992, data source = EPSG:3857 → geometry transformed to WGS84
90      SimpleFeatureSource featureSource = mockFeatureSource("EPSG:3857");
91  
92      String cql = "INTERSECTS(geometry, SRID=28992;" + RD_NL_POLYGON + ")";
93      Filter filter = FilterUtil.parseFilter(cql, RD_NL_APPLICATION, featureSource);
94  
95      assertNotNull(filter);
96      assertInstanceOf(Intersects.class, filter);
97  
98      Geometry transformedGeom =
99          (Geometry) ((Intersects) filter).getExpression2().evaluate(null);
100     assertNotNull(transformedGeom);
101     // Netherlands in EPSG:3857:
102     // Minimum Bounds (South-West Corner): X: 388,275.01, Y: 6,594,835.97
103     // Maximum Bounds (North-East Corner): X: 775,208.11, Y: 7,025,835.01
104     double x = transformedGeom.getCentroid().getX();
105     double y = transformedGeom.getCentroid().getY();
106     assertThat(
107         "Longitude should be in Netherlands EPSG:3857 range, was: " + x,
108         x,
109         is(both(greaterThan(388275d)).and(lessThan(775208d))));
110     assertThat(
111         "Latitude should be in Netherlands EPSG:3857 range, was: " + y,
112         y,
113         is(both(greaterThan(6594835d)).and(lessThan(7025835d))));
114   }

116   @Test
117   void spatial_filter_srid_does_not_match_app_crs_throws() {
118     // filter SRID = EPSG:3857, app CRS = EPSG:28992 → must throw
119     String cql = "INTERSECTS(geometry, SRID=3857;POLYGON((3.3 51.5, 7.2 51.5, 7.2 53.5, 3.3 53.5, 3.3 51.5)))";
120     assertThrows(
121         UnsupportedOperationException.class,
122         () -> FilterUtil.parseFilter(cql, RD_NL_APPLICATION, RDFeatureSource));
123   }
124 
125   @Test
126   void multiple_different_srids_in_filter_throws() {
127     String cql =
128         "INTERSECTS(geometry, SRID=28992;POINT(100000 400000)) OR INTERSECTS(geometry, SRID=3857;POINT(624339 6822408))";
129     assertThrows(
130         UnsupportedOperationException.class,
131         () -> FilterUtil.parseFilter(cql, RD_NL_APPLICATION, RDFeatureSource));
132   }
133 
134   @Test
135   void multiple_same_srids_in_filter_succeeds() throws CQLException, FactoryException {
136     // Two INTERSECTS with identical SRIDs → no exception
137     String cql =
138         "INTERSECTS(geometry, SRID=28992;POINT(100000 400000)) OR INTERSECTS(geometry, SRID=28992;POINT(200000 500000))";
139 
140     Filter filter = FilterUtil.parseFilter(cql, RD_NL_APPLICATION, RDFeatureSource);
141     assertNotNull(filter);
142     assertInstanceOf(Or.class, filter, "Filter should be an Or filter");
143   }
144 
145   @Test
146   void invalid_cql_throws_cql_exception() {
147     assertThrows(
148         CQLException.class,
149         () -> FilterUtil.parseFilter("NOT VALID CQL !!!", RD_NL_APPLICATION, RDFeatureSource));
150   }
