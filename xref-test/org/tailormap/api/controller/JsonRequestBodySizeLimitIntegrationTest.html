30  /**
31   * Test for the {@code spring.jackson.factory.constraints.read.max-document-length} limit on JSON request bodies.
32   *
33   * <p>This test does not use {@link org.tailormap.api.annotation.PostgresIntegrationTest} because that annotation
34   * includes {@code @EnableWebMvc}, which disables the Spring Boot WebMvc auto-configuration that configures the HTTP
35   * message converters with the Spring Boot configured {@code JsonMapper}. This test uses the same WebMvc configuration
36   * as the application.
37   */
38  @SpringBootTest(
39      classes = {
40        JPAConfiguration.class,
41        DataSourceAutoConfiguration.class,
42        HibernateJpaAutoConfiguration.class,
43        MailSenderAutoConfiguration.class,
44        ActuatorSecurityConfiguration.class,
45        ApiSecurityConfiguration.class,
46      })
47  @ComponentScan(basePackages = {"org.tailormap.api"})
48  @ActiveProfiles("postgresql")
49  @AutoConfigureMockMvc
50  class JsonRequestBodySizeLimitIntegrationTest {
51    private static final int MAX_DOCUMENT_LENGTH = 15 * 1024 * 1024;
52  
53    @Autowired
54    private MockMvc mockMvc;
55  
56    @Value("${tailormap-api.base-path}")
57    private String apiBasePath;
58  
59    @Test
60    @WithMockUser(
61        username = "tm-admin",
62        authorities = {ADMIN})
63    void drawing_exceeding_max_document_length_is_rejected() throws Exception {
64      // the description is well below the max string length, but the document exceeds the max document length
65      final String drawing = """
66  {"name":"Too large drawing","description":"%s","domainData":{},"featureCollection":null}
67  """.formatted("x".repeat(MAX_DOCUMENT_LENGTH));
68  
69      final String url = apiBasePath + "/app/default/drawing";
70      mockMvc.perform(put(url).accept(MediaType.APPLICATION_JSON)
71              .contentType(MediaType.APPLICATION_JSON)
72              .content(drawing)
73              .with(csrf()))
74          .andExpect(status().isBadRequest());
75    }
76  }
