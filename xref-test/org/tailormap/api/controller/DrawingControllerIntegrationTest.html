233   @Test
234   @Order(30)
235   @WithMockUser(
236       username = "tm-admin",
237       authorities = {ADMIN})
238   void update_drawing_by_admin() throws Exception {
239     String url = apiBasePath + "/drawing/list";
240     MvcResult result = mockMvc.perform(
241             get(url).accept(MediaType.APPLICATION_JSON).with(setServletPath(url)))
242         .andExpect(status().is2xxSuccessful())
243         .andExpect(status().is(200))
244         .andReturn();
245 
246     final String body = result.getResponse().getContentAsString();
247     assertNotNull(body, "response body should not be null");
248 
249     Drawing[] drawings = new JsonMapper().readValue(body, Drawing[].class);
250     // Select last drawing in collection
251     Drawing drawing = drawings.length == 0 ? null : drawings[drawings.length - 1];
252     assertNotNull(drawing, "drawing should not be null");
253 
254     String drawingId = drawing.getId().toString();
255     assertNotEquals(KNOWN_DRAWING_ID, drawingId);
256     Integer oldVersion = drawing.getVersion();
257     String oldName = drawing.getName();
258 
259     // update the NEW_DRAWING_JSON drawing, createdBy and createdAt are read-only and should be ignored
260     String updatedDrawing = NEW_DRAWING_JSON
261         .replace("Drawing 1 description", "Edited drawing 2 description")
262         .replace("\"name\":", "\"id\":\"" + drawingId + "\",\"name\":")
263         .replace("\"name\":", "\"createdBy\":\"forged-user\",\"name\":")
264         .replace("\"name\":", "\"createdAt\":\"2000-01-01T00:00:00Z\",\"name\":")
265         .replace("\"name\":", "\"version\":" + oldVersion + ",\"name\":");
266 
267     url = apiBasePath + "/app/default/drawing";
268     mockMvc.perform(put(url).accept(MediaType.APPLICATION_JSON)
269             .with(setServletPath(url))
270             .contentType(MediaType.APPLICATION_JSON)
271             .characterEncoding(Charset.defaultCharset())
272             .content(updatedDrawing))
273         .andExpect(status().isOk())
274         .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_JSON))
275         // API-created values
276         .andExpect(jsonPath("$.id").value(drawingId))
277         .andExpect(jsonPath("$.createdBy").value("tm-admin"))
278         .andExpect(jsonPath("$.createdAt", not(startsWith("2000-01-01"))))
279         .andExpect(jsonPath("$.updatedBy").value("tm-admin"))
280         .andExpect(jsonPath("$.srid").value(28992))
281         .andExpect(jsonPath("$.version").value(oldVersion + 1))
282         // given values
283         .andExpect(jsonPath("$.name").value(oldName))
284         .andExpect(jsonPath("$.description").value("Edited drawing 2 description"))
285         .andExpect(jsonPath("$.domainData").exists())
286         .andExpect(jsonPath("$.domainData.items").value(1))
287         .andExpect(jsonPath("$.domainData.domain").value("test drawings"))
288         .andExpect(jsonPath("$.featureCollection.features[0].geometry.type")
289             .value("Polygon"))
290         .andExpect(jsonPath("$.featureCollection.features[0].id", matchesPattern(UUID_REGEX)))
291         .andExpect(jsonPath("$.featureCollection.features[0].properties.prop0")
292             .value("value0"))
293         .andExpect(jsonPath("$.featureCollection.features[1].properties.rendering.fill")
294             .value("red"));
295   }

610   /**
611    * Authenticate as a non-owner user with the given additional property set to true. The test security context is
612    * cleared after each test.
613    */
614   private static void authenticateAsUserWithProperty(String propertyKey) {
615     TailormapUserDetails principal = new TailormapUserDetails() {
616       @Override
617       public Collection<? extends GrantedAuthority> getAuthorities() {
618         return List.of(new SimpleGrantedAuthority("test-bar"));
619       }
620 
621       @Override
622       public String getPassword() {
623         return "";
624       }
625 
626       @Override
627       public String getUsername() {
628         return "user";
629       }
630 
631       @Override
632       public String getOrganisation() {
633         return null;
634       }
635 
636       @Override
637       public Collection<TailormapAdditionalProperty> getAdditionalGroupProperties() {
638         return List.of(new TailormapAdditionalProperty(propertyKey, false, true));
639       }
640     };
641     TestSecurityContextHolder.setAuthentication(new UsernamePasswordAuthenticationToken(
642         principal, principal.getPassword(), principal.getAuthorities()));
643   }

645   @Test
646   @WithMockUser(
647       username = "tm-admin",
648       authorities = {ADMIN})
649   @Order(31)
650   void update_drawing_with_wrong_version_by_admin() throws Exception {
651     String url = apiBasePath + "/drawing/list";
652     MvcResult result = mockMvc.perform(
653             get(url).accept(MediaType.APPLICATION_JSON).with(setServletPath(url)))
654         .andExpect(status().isOk())
655         .andReturn();
656 
657     Drawing[] drawings = new JsonMapper().readValue(result.getResponse().getContentAsString(), Drawing[].class);
658     // Select last drawing in collection
659     Drawing drawing = drawings.length == 0 ? null : drawings[drawings.length - 1];
660     assertNotNull(drawing, "drawing should not be null");
661     assertNotEquals(KNOWN_DRAWING_ID, drawing.getId().toString());
662 
663     final String getUrl = apiBasePath + "/app/default/drawing/" + drawing.getId();
664     final String putUrl = apiBasePath + "/app/default/drawing";
665 
666     // both an outdated and a higher than current version must be rejected
667     for (int version : new int[] {drawing.getVersion() - 1, drawing.getVersion() + 1}) {
668       String updatedDrawing = NEW_DRAWING_JSON
669           .replace("Drawing 1 description", "Should not be stored")
670           .replace("\"name\":", "\"id\":\"" + drawing.getId() + "\",\"name\":")
671           .replace("\"name\":", "\"version\":" + version + ",\"name\":");
672 
673       mockMvc.perform(put(putUrl)
674               .accept(MediaType.APPLICATION_JSON)
675               .with(setServletPath(putUrl))
676               .contentType(MediaType.APPLICATION_JSON)
677               .characterEncoding(Charset.defaultCharset())
678               .content(updatedDrawing))
679           .andExpect(status().isConflict())
680           .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_JSON))
681           .andExpect(jsonPath("$.code").value(409))
682           .andExpect(jsonPath("$.message").value("Drawing has been updated by another user"));
683     }
684 
685     mockMvc.perform(get(getUrl).accept(MediaType.APPLICATION_JSON).with(setServletPath(getUrl)))
686         .andExpect(status().isOk())
687         .andExpect(jsonPath("$.version").value(drawing.getVersion()))
688         .andExpect(jsonPath("$.description").value(drawing.getDescription()));
689   }
