51    @Value("${tailormap-api.admin.base-path}")
52    private String adminBasePath;

54    private String getFeatureSourcePOSTBody(int port, String host, String database, String user, String password)
55        throws Exception {
56      return new ObjectMapper()
57          .writeValueAsString(Map.of(
58              "title", "My Test Source",
59              "protocol", "JDBC",
60              "url", "",
61              "jdbcConnection",
62                  Map.of(
63                      "dbtype", "postgis",
64                      "port", port,
65                      "host", host,
66                      "database", database,
67                      "schema", "public"),
68              "authentication",
69                  Map.of(
70                      "method", "password",
71                      "username", user,
72                      "password", password)));
73    }
74  
75    @Test
76    @WithMockUser(
77        username = "admin",
78        authorities = {Group.ADMIN})
79    void create_and_refresh_jdbc_feature_source_capabilities() throws Exception {
80      MockMvc mockMvc = MockMvcBuilders.webAppContextSetup(context).build(); // Required for Spring Data Rest APIs
81      final int expectedTotal = 39;
82  
83      String host = "localhost";
84      int port = 54322;
85      String database = "geodata";
86      String user = "geodata";
87      String password = "980f1c8A-25933b2";
88  
89      DriverManagerDataSource dataSource = new DriverManagerDataSource();
90      dataSource.setDriverClassName("org.postgresql.Driver");
91      dataSource.setUrl("jdbc:postgresql://%s:%s/%s".formatted(host, port, database));
92      dataSource.setUsername(user);
93      dataSource.setPassword(password);
94      JdbcTemplate jdbcTemplate = new JdbcTemplate(dataSource);
95      jdbcTemplate.execute("drop table if exists test");
96  
97      MvcResult result = mockMvc.perform(post(adminBasePath + "/feature-sources/new")
98              .param("catalogNodeId", "FeatureSourceAdminController")
99              .contentType(MediaType.APPLICATION_JSON)
100             .content(getFeatureSourcePOSTBody(port, host, database, user, password)))
101         .andExpect(status().isCreated())
102         .andExpect(header().string("Location", notNullValue()))
103         .andReturn();
104 
105     assertNotNull(result.getResponse().getRedirectedUrl());
106     String fsId = result.getResponse()
107         .getRedirectedUrl()
108         .substring(result.getResponse().getRedirectedUrl().lastIndexOf("/") + 1);
109 
110     // check that the created service is added to the catalog
111     Catalog catalog = catalogRepository.findById(Catalog.MAIN).orElseThrow();
112     CatalogNode attachedTo = catalog.getNodes().stream()
113         .filter(node -> node.getId().equals("FeatureSourceAdminController"))
114         .findFirst()
115         .orElseThrow();
116 
117     TailormapObjectRef objectRef = attachedTo.getItems().stream()
118         .filter(item -> fsId.equals(item.getId()))
119         .findFirst()
120         .orElseThrow();
121 
122     assertEquals(fsId, objectRef.getId());
123     assertEquals(TailormapObjectRef.KindEnum.FEATURE_SOURCE, objectRef.getKind());
124 
125     result = mockMvc.perform(get(result.getResponse().getRedirectedUrl()))
126         .andExpect(status().isOk())
127         .andExpect(jsonPath("$.id").isNotEmpty())
128         .andExpect(jsonPath("$.allFeatureTypes").isArray())
129         .andExpect(jsonPath("$.allFeatureTypes.length()").value(expectedTotal))
130         .andExpect(jsonPath("$.protocol").value("JDBC"))
131         .andReturn();
132 
133     Integer featureSourceId = JsonPath.read(result.getResponse().getContentAsString(), "$.id");
134     String selfLink = JsonPath.read(result.getResponse().getContentAsString(), "$._links.self.href");
135 
136     try {
137       jdbcTemplate.execute("create table test(id serial primary key)");
138 
139       mockMvc.perform(post(adminBasePath + "/feature-sources/%s/refresh-capabilities".formatted(featureSourceId)))
140           .andExpect(status().isFound())
141           .andExpect(header().string("Location", equalTo(selfLink)));
142 
143       mockMvc.perform(get(selfLink).accept(MediaType.APPLICATION_JSON))
144           .andExpect(status().isOk())
145           .andExpect(jsonPath("$.id").isNotEmpty())
146           .andExpect(jsonPath("$.allFeatureTypes").isArray())
147           .andExpect(jsonPath("$.allFeatureTypes.length()").value(expectedTotal + 1))
148           .andExpect(jsonPath("$.protocol").value("JDBC"))
149           .andExpect(jsonPath("$.allFeatureTypes[?(@.name=='test')]").isNotEmpty());
150     } finally {
151       try {
152         new JdbcTemplate(dataSource).execute("drop table test");
153       } catch (Exception e) {
154         // ignore
155       }
156     }
157 
158     mockMvc.perform(post(adminBasePath + "/feature-sources/%s/refresh-capabilities".formatted(featureSourceId)))
159         .andExpect(status().isFound())
160         .andExpect(header().string("Location", equalTo(selfLink)));
161 
162     mockMvc.perform(get(selfLink).accept(MediaType.APPLICATION_JSON))
163         .andExpect(status().isOk())
164         .andExpect(jsonPath("$.id").isNotEmpty())
165         .andExpect(jsonPath("$.allFeatureTypes").isArray())
166         .andExpect(jsonPath("$.allFeatureTypes.length()").value(expectedTotal))
167         .andExpect(jsonPath("$.protocol").value("JDBC"))
168         .andExpect(jsonPath("$.allFeatureTypes[?(@.name=='test')]").isEmpty());
169   }
