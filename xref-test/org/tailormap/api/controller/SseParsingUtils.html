6   package org.tailormap.api.controller;
7   
8   import java.util.Arrays;
9   import java.util.List;
10  import java.util.stream.Stream;
11  import org.tailormap.api.viewer.model.ServerSentEventResponse.EventTypeEnum;
12  import tools.jackson.core.JacksonException;
13  import tools.jackson.databind.JsonNode;
14  import tools.jackson.databind.json.JsonMapper;
15  
16  abstract class SseParsingUtils {
17  
18    /**
19     * Parse the {@code data:} lines from an SSE stream as JSON events. Each event looks something like:
20     * {@code {"details":{"message":"Extract task
21     * completed","progress":100,"downloadId":"begroeidterreindeel15061479295163305053.csv"},"eventType":"extract-completed","id":"019d6838-7f48-7053-9256-dd4b57c14264"}
22     * }
23     *
24     * <p>The stream may be read while the server is still writing to it, so a trailing line without a line terminator
25     * can be an unfinished message. Such a line is left unparsed.
26     */
27    private static Stream<JsonNode> parseEvents(String sseMessages) {
28      final int lastLineEnd = Math.max(sseMessages.lastIndexOf('\n'), sseMessages.lastIndexOf('\r'));
29      final String completeLines = sseMessages.substring(0, lastLineEnd + 1);
30      return Arrays.stream(completeLines.split("\\R"))
31          .map(String::trim)
32          .filter(line -> line.startsWith("data:"))
33          .map(line -> JsonMapper.shared().readTree(line.substring("data:".length())));
34    }
35  
36    private static Stream<JsonNode> parseEvents(String sseMessages, EventTypeEnum eventType) {
37      return parseEvents(sseMessages)
38          .filter(event ->
39              eventType.getValue().equals(event.path("eventType").asString()));
40    }
41  
42    /** Get the last extract completed event from the SSE stream as JSON. */
43    String getLastCompletedEventJson(String sseMessages) {
44      List<JsonNode> completedEvents =
45          parseEvents(sseMessages, EventTypeEnum.EXTRACT_COMPLETED).toList();
46      // Select last completed event, throws NoSuchElementException if there is none
47      return completedEvents.getLast().toString();
48    }
49  
50    String getDownloadId(String eventJson) throws JacksonException {
51      return JsonMapper.shared()
52          .readTree(eventJson)
53          .path("details")
54          .path("downloadId")
55          .asString();
56    }
57  
58    int count_completed_messages(String stream) {
59      return (int) parseEvents(stream, EventTypeEnum.EXTRACT_COMPLETED).count();
60    }
61  
62    int count_all_keep_alive_messages(String stream) {
63      return (int) parseEvents(stream, EventTypeEnum.KEEP_ALIVE).count();
64    }
65  
66    /** Count the keep-alive messages of the viewer event bus, which have empty details and an id. */
67    int count_viewer_keep_alive_messages(String stream) {
68      return (int) parseEvents(stream, EventTypeEnum.KEEP_ALIVE)
69          .filter(event -> event.path("details").isObject()
70              && event.path("details").isEmpty()
71              && event.has("id"))
72          .count();
73    }
74  
75    /** Count the keep-alive messages of the admin event bus, which have null details. */
76    int count_admin_keep_alive_messages(String stream) {
77      return (int) parseEvents(stream, EventTypeEnum.KEEP_ALIVE)
78          .filter(event -> event.has("details") && event.get("details").isNull())
79          .count();
80    }
81  }
