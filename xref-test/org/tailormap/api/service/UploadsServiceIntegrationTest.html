6   package org.tailormap.api.service;
7   
8   import static org.junit.jupiter.api.Assertions.assertFalse;
9   import static org.junit.jupiter.api.Assertions.assertTrue;
10  
11  import java.time.OffsetDateTime;
12  import java.time.ZoneId;
13  import java.util.UUID;
14  import org.junit.jupiter.api.BeforeAll;
15  import org.junit.jupiter.api.Test;
16  import org.springframework.beans.factory.annotation.Autowired;
17  import org.tailormap.api.annotation.PostgresIntegrationTest;
18  import org.tailormap.api.persistence.Upload;
19  import org.tailormap.api.persistence.UploadCategory;
20  import org.tailormap.api.repository.UploadRepository;
21  
22  @PostgresIntegrationTest
23  class UploadsServiceIntegrationTest {
24    private static OffsetDateTime now;
25  
26    @Autowired
27    private UploadsService uploadsService;
28  
29    @Autowired
30    private UploadRepository uploadRepository;
31  
32    @BeforeAll
33    static void setup() {
34      now = OffsetDateTime.now(ZoneId.systemDefault());
35    }
36  
37    @Test
38    void check_if_modified_since_two_weeks_ago() {
39      Upload logo = uploadRepository.findByCategory(UploadCategory.APP_LOGO).getFirst();
40  
41      long twoWeeksBeforeNow = now.minusWeeks(2).toInstant().toEpochMilli();
42  
43      assertTrue(uploadsService.checkIfModifiedSince(logo.getId(), twoWeeksBeforeNow));
44    }
45  
46    @Test
47    void check_if_modified_since_a_week_later() {
48      Upload logo = uploadRepository.findByCategory(UploadCategory.APP_LOGO).getFirst();
49  
50      long aWeekInTheFuture = now.plusWeeks(1).toInstant().toEpochMilli();
51  
52      assertFalse(uploadsService.checkIfModifiedSince(logo.getId(), aWeekInTheFuture));
53    }
54  
55    @Test
56    void check_if_modified_since_not_found() {
57      assertTrue(
58          uploadsService.checkIfModifiedSince(UUID.randomUUID(), System.currentTimeMillis()),
59          "Should return true for non-existing upload");
60    }
61  }
