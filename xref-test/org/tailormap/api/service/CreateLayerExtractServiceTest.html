32  class CreateLayerExtractServiceTest {
33    @TempDir
34    private Path tempDir;
35  
36    /**
37     * The extract is created asynchronously, so any exception must be reported to the client with an SSE event,
38     * otherwise the client waits indefinitely for the extract.
39     */
40    @ParameterizedTest
41    @EnumSource(ExtractOutputFormat.class)
42    @SuppressWarnings("PMD.DisposeDataStore") // openGeoToolsFeatureSource() is only stubbed, nothing to dispose
43    void should_emit_error_for_any_exception(ExtractOutputFormat outputFormat) throws Exception {
44      SseEventBus eventBus = mock(SseEventBus.class);
45      FeatureSourceFactoryHelper featureSourceFactoryHelper = mock(FeatureSourceFactoryHelper.class);
46      when(featureSourceFactoryHelper.openGeoToolsFeatureSource(any(TMFeatureType.class)))
47          .thenThrow(new IllegalStateException("Something unexpected went wrong"));
48  
49      CreateLayerExtractService service = new CreateLayerExtractService(
50          eventBus, JsonMapper.builder().build(), featureSourceFactoryHelper, mock(ZipService.class));
51      ReflectionTestUtils.setField(service, "exportFilesBaseLocation", tempDir.toString());
52      ReflectionTestUtils.setField(service, "progressReportInterval", 100);
53      service.initializeExtractDirectory();
54  
55      service.createLayerExtract(
56          "test-client",
57          new TMFeatureType().setName("test_feature_type"),
58          Set.of("naam"),
59          null,
60          null,
61          null,
62          outputFormat,
63          "test-extract" + outputFormat.getExtension());
64  
65      ArgumentCaptor<SseEvent> events = ArgumentCaptor.forClass(SseEvent.class);
66      verify(eventBus, atLeast(2)).handleEvent(events.capture());
67      List<SseEvent> allEvents = events.getAllValues();
68      String lastEventData = String.valueOf(allEvents.getLast().data());
69      assertThat(lastEventData, containsString("extract-failed"));
70      assertThat(lastEventData, containsString("Something unexpected went wrong"));
71    }
72  }
