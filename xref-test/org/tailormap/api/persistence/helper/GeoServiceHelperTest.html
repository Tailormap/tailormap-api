7   package org.tailormap.api.persistence.helper;
8   
9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  
11  import java.net.URI;
12  import java.util.ArrayList;
13  import java.util.List;
14  import org.junit.jupiter.api.Test;
15  import org.tailormap.api.persistence.GeoService;
16  import org.tailormap.api.persistence.json.GeoServiceLayer;
17  import org.tailormap.api.persistence.json.WMSStyle;
18  
19  class GeoServiceHelperTest {
20  
21    @Test
22    void get_layer_legend_url_from_styles_does_not_modify_layer() {
23      WMSStyle parentStyle = new WMSStyle().name("parent").legendUrl(URI.create("http://example.com/parent"));
24      WMSStyle childStyle = new WMSStyle().name("child").legendUrl(URI.create("http://example.com/child"));
25  
26      GeoServiceLayer parent = new GeoServiceLayer()
27          .id("0")
28          .name("parent")
29          .root(true)
30          .styles(new ArrayList<>(List.of(parentStyle)))
31          .children(new ArrayList<>(List.of("1")));
32      // Child layer inherits the parent style, as in GeoServer capabilities
33      GeoServiceLayer child = new GeoServiceLayer()
34          .id("1")
35          .name("child")
36          .root(false)
37          .styles(new ArrayList<>(List.of(parentStyle, childStyle)))
38          .children(new ArrayList<>());
39  
40      GeoService service = new GeoService().setLayers(new ArrayList<>(List.of(parent, child)));
41  
42      int hashCodeBefore = child.hashCode();
43  
44      URI legendUrl = GeoServiceHelper.getLayerLegendUrlFromStyles(service, child);
45  
46      assertEquals(URI.create("http://example.com/child"), legendUrl);
47      assertEquals(List.of(parentStyle, childStyle), child.getStyles(), "layer styles must not be modified");
48      assertEquals(hashCodeBefore, child.hashCode(), "layer hashCode must not change");
49    }
50  }
