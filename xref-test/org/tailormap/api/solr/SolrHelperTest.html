30    @Test
31    void find_in_index_transforms_spatial_query_and_result_geometry() throws Exception {
32      SolrClient solrClient = mock(SolrClient.class);
33      QueryResponse queryResponse = mock(QueryResponse.class);
34  
35      String sourceGeometryWkt = "POINT (3.7651071548461914 51.742008209228516)";
36  
37      SolrDocument document = new SolrDocument();
38      document.setField("id", "lgstations.108");
39      document.setField("displayFields", List.of("Jan van Renesseweg"));
40      document.setField("geometry", sourceGeometryWkt);
41  
42      SolrDocumentList documents = new SolrDocumentList();
43      documents.add(document);
44      documents.setNumFound(1);
45      documents.setStart(0);
46  
47      when(queryResponse.getResults()).thenReturn(documents);
48      when(solrClient.query(any(SolrQuery.class))).thenReturn(queryResponse);
49  
50      SearchIndex searchIndex =
51          new SearchIndex().setId(7L).setName("Reddingposten").setSourceCrs("EPSG:4326");
52  
53      Geometry sourcePoint = GeometryProcessor.wktToGeometry(sourceGeometryWkt);
54      assertNotNull(sourcePoint);
55  
56      /*
57       * Known EPSG:3857 coordinates for longitude 3.7651071548461914,
58       * latitude 51.742008209228516. Keep these values independent from
59       * TransformationUtil so axis-order regressions are detected.
60       */
61      String applicationPoint = "419129.8112595891 6753610.983561384";
62      Geometry applicationPointGeometry = GeometryProcessor.wktToGeometry("POINT (" + applicationPoint + ")");
63      assertNotNull(applicationPointGeometry);
64  
65      double distance = 0.005;
66  
67      SearchResponse response;
68      try (SolrHelper solrHelper = new SolrHelper(solrClient)) {
69        response = solrHelper.findInIndex(searchIndex, "*", null, applicationPoint, distance, 0, 10, "EPSG:3857");
70      }
71  
72      ArgumentCaptor<SolrQuery> queryCaptor = ArgumentCaptor.forClass(SolrQuery.class);
73      verify(solrClient).query(queryCaptor.capture());
74  
75      SolrQuery sentQuery = queryCaptor.getValue();
76  
77      /*
78       * The point supplied by the application must be transformed back to the
79       * source CRS before it is sent to Solr.
80       */
81      String transformedPoint = sentQuery.get("pt");
82      assertNotNull(transformedPoint);
83  
84      Geometry transformedPointGeometry = GeometryProcessor.wktToGeometry("POINT (" + transformedPoint + ")");
85  
86      assertNotNull(transformedPointGeometry);
87      assertEquals(sourcePoint.getCoordinate().x, transformedPointGeometry.getCoordinate().x, 0.000001);
88      assertEquals(sourcePoint.getCoordinate().y, transformedPointGeometry.getCoordinate().y, 0.000001);
89  
90      /*
91       * Solr distance is already expressed in its configured distance unit and
92       * must not be transformed together with the point coordinates.
93       */
94      assertEquals("0.005", sentQuery.get("d"));
95  
96      assertEquals(1, response.getDocuments().size());
97  
98      String geometry = response.getDocuments().getFirst().getGeometry();
99      assertNotNull(geometry);
100 
101     /*
102      * Geometry stored in Solr uses the source CRS. The geometry returned by
103      * the API must use the application CRS.
104      */
105     Geometry transformedGeometry = GeometryProcessor.wktToGeometry(geometry);
106     assertNotNull(transformedGeometry);
107 
108     assertEquals(applicationPointGeometry.getCoordinate().x, transformedGeometry.getCoordinate().x, 0.001);
109     assertEquals(applicationPointGeometry.getCoordinate().y, transformedGeometry.getCoordinate().y, 0.001);
110   }

155   @Test
156   void find_in_index_without_source_crs_preserves_existing_behavior() throws Exception {
157     SolrClient solrClient = mock(SolrClient.class);
158     QueryResponse queryResponse = mock(QueryResponse.class);
159 
160     String geometryWkt = "POINT (3.7651071548461914 51.742008209228516)";
161 
162     SolrDocument document = new SolrDocument();
163     document.setField("id", "lgstations.108");
164     document.setField("displayFields", List.of("Jan van Renesseweg"));
165     document.setField("geometry", geometryWkt);
166 
167     SolrDocumentList documents = new SolrDocumentList();
168     documents.add(document);
169     documents.setNumFound(1);
170     documents.setStart(0);
171 
172     when(queryResponse.getResults()).thenReturn(documents);
173     when(solrClient.query(any(SolrQuery.class))).thenReturn(queryResponse);
174 
175     /*
176      * Existing search_index records will initially have a null sourceCrs after
177      * the database migration, until the index has been rebuilt.
178      */
179     SearchIndex searchIndex = new SearchIndex().setId(7L).setName("Existing search index");
180 
181     String point = "419121.8 6750463.6";
182     double distance = 0.005;
183 
184     SearchResponse response;
185     try (SolrHelper solrHelper = new SolrHelper(solrClient)) {
186       response = solrHelper.findInIndex(searchIndex, "*", null, point, distance, 0, 10, "EPSG:3857");
187     }
188 
189     ArgumentCaptor<SolrQuery> queryCaptor = ArgumentCaptor.forClass(SolrQuery.class);
190     verify(solrClient).query(queryCaptor.capture());
191 
192     SolrQuery sentQuery = queryCaptor.getValue();
193 
194     /*
195      * Without a stored source CRS Tailormap cannot safely transform either the
196      * spatial query point or the returned geometry, so the previous behaviour
197      * is retained.
198      */
199     assertEquals(point, sentQuery.get("pt"));
200     assertEquals("0.005", sentQuery.get("d"));
201 
202     assertEquals(1, response.getDocuments().size());
203     assertEquals(geometryWkt, response.getDocuments().getFirst().getGeometry());
204   }
