6   package org.tailormap.api.configuration;
7   
8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   
10  import java.sql.Date;
11  import java.sql.Timestamp;
12  import java.time.Instant;
13  import java.util.LinkedHashMap;
14  import java.util.Map;
15  import org.junit.jupiter.api.Test;
16  import org.springframework.beans.factory.annotation.Autowired;
17  import org.springframework.boot.test.autoconfigure.json.JsonTest;
18  import org.springframework.context.annotation.Import;
19  import org.tailormap.api.viewer.model.Feature;
20  import tools.jackson.databind.json.JsonMapper;
21  
22  /**
23   * Checks that the auto-configured JsonMapper (also used by Spring MVC) serializes {@link java.sql.Date} values as a
24   * date without a time. The application picks up {@link SqlDateSerializer} with the component scan in
25   * {@code TailormapApiConfiguration}, which the {@code @JsonTest} slice does not run, so it is imported here.
26   */
27  @JsonTest
28  @Import(SqlDateSerializer.class)
29  class SqlDateSerializerTest {
30    @Autowired
31    private JsonMapper jsonMapper;
32  
33    @Test
34    void sql_date_is_serialized_as_date_only() {
35      assertEquals("\"2026-10-21\"", jsonMapper.writeValueAsString(Date.valueOf("2026-10-21")));
36    }
37  
38    @Test
39    void sql_date_in_feature_attributes_is_serialized_as_date_only() {
40      // Feature attributes are a Map<String, Object> filled with the values GeoTools returns
41      Map<String, Object> attributes = new LinkedHashMap<>();
42      attributes.put("creationdate", Date.valueOf("2026-10-21"));
43      attributes.put("terminationdate", null);
44      Feature feature = new Feature().fid("test.1").attributes(attributes);
45  
46      String json = jsonMapper.writeValueAsString(feature);
47  
48      assertEquals(
49          "2026-10-21",
50          jsonMapper.readTree(json).get("attributes").get("creationdate").asString());
51    }
52  
53    @Test
54    void timestamp_is_still_serialized_as_timestamp() {
55      String json = jsonMapper.writeValueAsString(Timestamp.from(Instant.parse("2026-10-21T10:15:30Z")));
56      assertEquals("\"2026-10-21T10:15:30.000Z\"", json);
57    }
58  }
