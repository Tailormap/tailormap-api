6   package org.tailormap.api.configuration;
7   
8   import static org.junit.jupiter.api.Assertions.assertDoesNotThrow;
9   import static org.junit.jupiter.api.Assertions.assertNotNull;
10  
11  import org.geotools.filter.FunctionFinder;
12  import org.junit.jupiter.api.Test;
13  
14  /**
15   * The class initialization deadlock {@link GeoToolsConfiguration#warmUpFactories()} prevents only happens with
16   * concurrent first use, which can't be reproduced reliably in a test. This only checks the warm-up itself works and
17   * leaves the function lookup working.
18   */
19  class GeoToolsConfigurationTest {
20    @Test
21    void warm_up_factories_loads_filter_functions() {
22      assertDoesNotThrow(GeoToolsConfiguration::warmUpFactories);
23      FunctionFinder functionFinder = new FunctionFinder(null);
24      // Function used by UniqueValuesHelper
25      assertNotNull(functionFinder.findFunctionDescription("Collection_Unique"));
26      // Function initialized by SQLDialect via InFunction
27      assertNotNull(functionFinder.findFunctionDescription("in"));
28    }
29  }
